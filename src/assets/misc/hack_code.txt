#define DEFAULT_PORT 4444

int create_socket(int port) {
    int sockfd;
    struct sockaddr_in server;
    sockfd = socket(AF_INET, SOCK_STREAM, 0);
    if (sockfd < 0) {
        perror("socket");
        exit(EXIT_FAILURE);
    }
    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY;
    server.sin_port = htons(port);
    if (bind(sockfd, (struct sockaddr *)&server, sizeof(server)) < 0) {
        perror("bind");
        exit(EXIT_FAILURE);
    }
    listen(sockfd, 5);
    return sockfd;
}

    int sockfd = create_socket(DEFAULT_PORT);
    int client;
    char buffer[BUFFER_SIZE];
    printf("[+] Listening on port %d\n", DEFAULT_PORT);

        client = accept(sockfd, NULL, NULL);
        read(client, buffer, BUFFER_SIZE);
        printf("[+] Received: %s\n", buffer);

    return 0;
}

/* ==== scanner.py ==== */
import socket
import sys
import time

def tcp_scan(target, ports):
    open_ports = []
    for port in ports:
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.settimeout(0.8)
        result = s.connect_ex((target, port))
        if result == 0:
            open_ports.append(port)
            print(f"[+] {target}:{port} OPEN")
        s.close()
    return open_ports

def main():
    target = sys.argv[1] if len(sys.argv) > 1 else "127.0.0.1"
    ports = [21, 22, 23, 80, 110, 443, 445, 1433, 3306, 3389, 5432, 8080]
    print(f"[*] Scanning {target} - {len(ports)} ports")
    found = tcp_scan(target, ports)
    print(f"[*] Scan complete: {len(found)} open ports")
    for p in found:
        print(f"    -> {target}:{p}")

if __name__ == "__main__":
    main()

class Unpacker
  def initialize(data)
    @data = data
  end

  def decrypt(key)
    result = ""
    @data.bytes.each_with_index do |byte, i|
      result << (byte ^ key.bytes[i % key.bytesize]).chr
    end
    result
  end

  def unpack(payload)
    magic = payload.bytes[0, 2].pack("C*")
    raise "bad magic" unless magic == "MZ"
    offset = payload.bytes[2, 2].pack("C*").unpack1("v")
    payload.byteslice(offset, payload.bytesize - offset)
  end
end
